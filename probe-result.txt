function go_to_player(encryptedUrl) {
    // Mostrar loading
    document.querySelector(".BotHumano")
        .classList.add('BotHumanoA');
    document.querySelector(".BotHumano")
        .insertAdjacentHTML('afterbegin', '<div class="BoxHumano"><span><b>Procesando vídeo</b> <p>Por favor espere unos segundos</p></span></div>');
    
    // Desencriptar URL usando API backend
    fetch('/embed/api/decrypt-stream', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify({ encrypted: encryptedUrl })
    })
    .then(response => response.json())
    .then(data => {
        if (data.error) {
            // Error al desencriptar (no mostrar detalles en console);
            document.querySelector(".BotHumano").classList.remove('BotHumanoA');
            alert('Error al cargar el video: ' + data.error);
            return;
        }
        
        // Cargar video con URL desencriptada
        loadVideoPlayer(data.url);
    })
    .catch(error => {
        console.error('Error en la petición:', error);
        document.querySelector(".BotHumano").classList.remove('BotHumanoA');
        alert('Error al cargar el video. Por favor, inténtalo de nuevo.');
    });
}

}
